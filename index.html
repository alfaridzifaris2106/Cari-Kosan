Kode Google Apps Script (Auto-Timestamp & Sync)

Tempel kode ini di Google Sheets Anda melalui menu Ekstensi > Apps Script lalu tekan Simpan. Untuk menerima update otomatis dari aplikasi web ini, klik Deploy > New deployment > Web app (akses: Anyone), lalu tempel URL-nya ke kolom Sinkronisasi Otomatis. 

function onEdit(e) {
  var sheet = e.source.getActiveSheet();
  if (sheet.getName() !== 'Database & Pelacak Survei Kosan') return;
  var row = e.range.getRow();
  var col = e.range.getColumn();
  if (row < 9 || col === 26) return;

  var now = Utilities.formatDate(new Date(), 'Asia/Jakarta', 'yyyy-MM-dd HH:mm:ss');
  sheet.getRange(row, 26).setValue(now);

  var logSheet = e.source.getSheetByName('Log Perubahan');
  if (logSheet) {
    var kode = sheet.getRange(row, 6).getValue();
    var nama = sheet.getRange(row, 8).getValue();
    var header = sheet.getRange(8, col).getValue();
    logSheet.insertRowAfter(1);
    logSheet.getRange(2, 1, 1, 6).setValues([[now, kode, nama, 'Spreadsheet', header, 'Nilai diubah menjadi: ' + e.range.getValue()]]);
  }
}

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('Database & Pelacak Survei Kosan');
  var rows = sheet.getDataRange().getDisplayValues();
  var formulas = sheet.getDataRange().getFormulas();
  var out = [];
  for (var r = 8; r < rows.length; r++) {
    var row = rows[r];
    if (!row[7]) continue;
    var fWa = formulas[r][14] || '';
    var fMaps = formulas[r][15] || '';
    var fDrive = formulas[r][24] || '';
    var exUrl = function(f, d) { var m = f.match(/"(https?:[^"]+)"/); return m ? m[1] : d; };
    out.push({
      kontak: row[1] === 'TRUE', survei: row[2] === 'TRUE', tempati: row[3] === 'TRUE',
      kode_input: row[4], kode: row[5], kecamatan: row[6], nama: row[7], alamat: row[8],
      hargaText: row[9], tipe: row[10], akses: row[11], fasilitas: row[12], transportasi: row[13],
      wa: exUrl(fWa, row[14]), wa_display: row[14], maps: exUrl(fMaps, row[15]),
      aturan: row[16], catatan: row[17], status: row[19],
      hargaMin: Number(String(row[20]).replace(/[^0-9]/g, '')) || 0,
      hargaMax: Number(String(row[21]).replace(/[^0-9]/g, '')) || 0,
      jadwal: row[22], skor: Number(row[23]) || 0,
      driveUrl: exUrl(fDrive, row[24]), lastUpdated: row[25]
    });
  }
  return ContentService.createTextOutput(JSON.stringify({ ok: true, items: out })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('Database & Pelacak Survei Kosan');
  var logSheet = ss.getSheetByName('Log Perubahan');
  var now = Utilities.formatDate(new Date(), 'Asia/Jakarta', 'yyyy-MM-dd HH:mm:ss');

  var names = sheet.getRange(9, 8, sheet.getLastRow() - 8, 1).getValues();
  var targetRow = -1;
  for (var i = 0; i < names.length; i++) {
    if (String(names[i][0]).trim().toLowerCase() === String(data.nama).trim().toLowerCase()) {
      targetRow = i + 9;
      break;
    }
  }
  if (targetRow === -1) {
    targetRow = sheet.getLastRow() + 1;
    sheet.getRange(targetRow, 5, 1, 4).setValues([[data.kode_input || 'Custom', data.kode, data.kecamatan, data.nama]]);
  }
  sheet.getRange(targetRow, 2, 1, 3).setValues([[!!data.kontak, !!data.survei, !!data.tempati]]);
  sheet.getRange(targetRow, 9).setValue(data.alamat);
  sheet.getRange(targetRow, 10).setValue(data.hargaText);
  sheet.getRange(targetRow, 13).setValue(data.fasilitas);
  sheet.getRange(targetRow, 20, 1, 3).setValues([[data.status, data.hargaMin, data.hargaMax]]);
  sheet.getRange(targetRow, 26).setValue(now);

  if (logSheet) {
    logSheet.insertRowAfter(1);
    logSheet.getRange(2, 1, 1, 6).setValues([[now, data.kode, data.nama, 'Aplikasi Web', data.changedFields || 'Detail Kosan', data.summary || 'Diperbarui dari aplikasi']]);
  }
  return ContentService.createTextOutput(JSON.stringify({ ok: true, row: targetRow })).setMimeType(ContentService.MimeType.JSON);
}
